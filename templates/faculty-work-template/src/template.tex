\newcommand{\makecover}{%
  \begin{titlepage}
    \thispagestyle{empty}

    \begin{tikzpicture}[remember picture,overlay]
      \fill[CoverAccent] (current page.north west) rectangle ([xshift=4.0mm]current page.south west);
    \end{tikzpicture}

    \vspace*{10mm}

    \begin{center}
      \IfFileExists{assets/\coverlogo}{%
        \includegraphics[width=0.78\textwidth]{\coverlogo}
      }{}
    \end{center}

    \vspace{12mm}

    \begin{center}
      {\bfseries\Large \university\par}
      \vspace{2mm}
      {\bfseries\normalsize \school\par}
      \vspace{1mm}
      {\normalsize \career\par}
    \end{center}

    \vspace{14mm}

    \noindent
    {\bfseries \subject}\par
    \vspace{2mm}
    {\bfseries \doctype\ \docid}\par

    \vspace{10mm}

    \noindent
    {\bfseries\fontsize{26}{32}\selectfont \titlee\par}

    \vspace{4mm}
    \noindent
    {\normalsize \subtitle\par}

    \vspace{8mm}
    \noindent{\color{CoverAccent}\rule{\textwidth}{1.0pt}}

  \end{titlepage}%
}
