\newcommand{\makecover}{%
  \begin{titlepage}
    \thispagestyle{empty}
    \color{Ink}

    \begin{tikzpicture}[remember picture,overlay]
      \fill[Burgundy] (current page.north west) rectangle ([xshift=3.2mm]current page.south west);
      \fill[Burgundy] ([xshift=3.2mm,yshift=-10mm]current page.north west)
        rectangle ([xshift=18mm,yshift=-12mm]current page.north west);
    \end{tikzpicture}

    \vspace*{10mm}

    \noindent
    {\bfseries\fontsize{12}{14}\selectfont \library}\hfill
    {\fontsize{10.5}{12}\selectfont \tagline}

    \vspace{8mm}
    \noindent{\color{Rule}\rule{\textwidth}{0.6pt}}
    \color{Ink}

    \vspace{18mm}

    {\raggedright\bfseries\fontsize{36}{40}\selectfont\hyphenchar\font=-1\relax \titlee\par}
    \vspace{5mm}
    {\raggedright\mdseries\fontsize{12.5}{16}\selectfont\hyphenchar\font=-1\relax \subtitle\par}

    \vspace{12mm}

    {\bfseries\fontsize{11}{13}\selectfont \category\par}
    \vspace{2.5mm}
    {\color{Burgundy}\rule{24mm}{1.2pt}}\par
    \color{Ink}

    \vfill

    \noindent
    \begin{tabularx}{\textwidth}{@{}lX@{}}
      {\bfseries\fontsize{9.5}{11}\selectfont \typeLabel} &
      {\mdseries\fontsize{9.5}{11}\selectfont \doctype} \\

      {\mdseries\fontsize{9.5}{11}\selectfont \version} \\
